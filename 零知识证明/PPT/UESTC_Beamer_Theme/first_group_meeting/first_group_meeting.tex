\begin{frame}{本次汇报想回答的问题}
  \begin{itemize}
    \item 零知识证明到底解决了什么信任问题？
    \item 近期阅读的论文分别推进了哪一类能力？
    \item 从论文中学到哪些可迁移的研究方法？
    \item 是否能提出一个足够小、可实验验证的新方向？
  \end{itemize}
  \vspace{0.4em}
  \begin{block}{汇报主线}
    从“证明一个结果正确”走向“证明一个 AI 系统按承诺、权限与策略执行”。
  \end{block}
\end{frame}

\begin{frame}{零知识证明：一句话理解}
  \begin{block}{基本形式}
    证明者向验证者证明存在秘密见证 $w$，使公开关系 $R(x,w)=1$ 成立，
    但不泄露 $w$ 本身。
  \end{block}
  \[
    \exists w:\ R(x,w)=1
  \]
  \begin{itemize}
    \item $x$：公开陈述，如承诺、输出、模型哈希、策略版本。
    \item $w$：秘密见证，如输入、模型参数、中间计算 trace。
    \item $\pi$：证明，验证者只需检查 $\mathrm{Verify}(x,\pi)=1$。
  \end{itemize}
\end{frame}

\begin{frame}{三条性质与能力边界}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{核心性质}
        \begin{itemize}
          \item \kw{完备性}：诚实执行应当通过验证。
          \item \kw{可靠性}：错误声明难以伪造证明。
          \item \kw{零知识性}：证明过程不泄露秘密。
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{需要保持清醒}
        \begin{itemize}
          \item 不证明输入数据真实。
          \item 不证明模型公平、无偏或无后门。
          \item 不证明需求和规则本身合理。
          \item 证明的是“按给定规则正确执行”。
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{DIZK：把证明生成从单机推向集群}
  \begin{itemize}
    \item 问题：传统 zkSNARK prover 是单机单体进程，内存随电路规模迅速膨胀。
    \item 方法：将证明生成拆分到计算集群，处理多项式、FFT、约束计算等并行任务。
    \item 结果：可扩展到十亿级逻辑门，论文报告比先前系统支持更大规模计算。
    \item 我的理解：ZKP 落地不只是协议正确，还要解决数据划分、通信、容错与资源调度。
  \end{itemize}
  \smallcite{Howard Wu et al., DIZK, USENIX Security 2018}
\end{frame}

\begin{frame}{zkCNN 与 Mystique：ZKML 的核心难点}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{zkCNN}
        \begin{itemize}
          \item 面向 CNN 预测和准确率证明。
          \item 针对卷积、ReLU、MaxPool 设计交互式证明。
          \item 启发：利用 ML 结构，而不是直接套通用电路。

      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Mystique}
        \begin{itemize}
          \item 关注算术/布尔、定点/浮点、承诺/认证值转换。
          \item 优化矩阵乘证明，面向大规模神经网络推理。
          \item 启发：转换层是系统性能的隐藏主战场。
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Kaizen 与 FAIRZK：从推理正确到训练/性质正确}
  \begin{itemize}
    \item \kw{Kaizen}：证明模型确实按指定训练算法和数据承诺完成训练；递归组合使证明大小和验证时间不随迭代数增长。
    \item \kw{FAIRZK}：证明模型满足公平性约束；关键不是逐样本证明每次推理，而是用模型参数与聚合信息构造公平性上界。
    \item 共同趋势：ZKP 正在从“证明一次计算输出”扩展到“证明模型来源、训练过程和高层性质”。
  \end{itemize}
  \begin{block}{对我的启发}
    好的 ZK 应用往往不是把原任务硬塞进电路，而是重新寻找可证明、可压缩、可验证的等价表述。
  \end{block}
\end{frame}

\begin{frame}{Agent 安全论文给 ZKP 的新入口}
  \begin{itemize}
    \item \kw{SAGA}：强调 Agent 身份、注册、访问控制 token、跨 Agent 通信策略。
    \item \kw{Cloak, Honey, Trap}：展示 LLM Agent 会被诱导、误导、循环和反制，安全边界非常脆弱。
    \item 共同问题：Agent 行为越来越自治，但用户和第三方很难确认它是否按权限、策略、版本执行。
  \end{itemize}
  \vspace{0.4em}
  \begin{block}{连接点}
    Agent 治理需要“可验证执行记录”；ZKP 可以让验证者相信合规性，而不暴露完整提示词、私有工具输入或业务数据。
  \end{block}
\end{frame}

\begin{frame}{阶段性心得：我学到的四件事}
  \begin{enumerate}
    \item \kw{先定义可验证关系}：论文的核心常常藏在 $R(x,w)$ 怎么选。
    \item \kw{避免天真电路化}：直接证明完整系统通常太慢，要找结构化捷径。
    \item \kw{区分证明目标}：推理正确、训练正确、公平性、权限合规是不同命题。
    \item \kw{工程指标同样重要}：证明时间、内存、证明大小、验证时间决定能否落地。
  \end{enumerate}
\end{frame}

\begin{frame}{可学习的方法论}
  \begin{itemize}
    \item 从应用痛点出发：谁不信任谁？秘密是什么？公开可验证的承诺是什么？
    \item 把系统拆成公开输入、秘密见证、中间 trace、验证关系四部分。
    \item 优先寻找“证明高层性质”的路径，而不是逐步证明每次底层计算。
    \item 做实验时先建最小原型：小模型、小策略、小数据集，跑通证明链路后再优化。
  \end{itemize}
\end{frame}

\begin{frame}{新的可落地 idea：ZK-AgentGuard}
  \begin{block}{一句话}
    为 LLM Agent 生成“权限合规证明”：证明它在一次任务中只调用了被授权工具、使用了指定策略版本、没有越权访问敏感资源，同时不公开完整上下文。
  \end{block}
  \begin{itemize}
    \item 适用场景：企业内部 Agent、科研助理 Agent、自动化运维 Agent。
    \item 公开输入：Agent 身份、策略哈希、工具白名单哈希、任务输出摘要。
    \item 秘密见证：完整调用日志、用户私有输入、工具返回内容、内部推理状态摘要。
  \end{itemize}
\end{frame}

\begin{frame}{可能的难点与应对}
  \begin{itemize}
    \item \kw{语义鸿沟}：自然语言策略难以直接证明。\\
      应对：先设计结构化策略 DSL，再由自然语言映射到 DSL。
    \item \kw{日志可信性}：Agent 可能不记录真实调用。\\
      应对：工具代理层强制记录并签名，Agent 只能通过代理调用工具。
    \item \kw{性能压力}：调用日志很长时约束数增长。\\
      应对：用 Merkle 承诺、分段证明、递归聚合降低验证成本。
    \item \kw{威胁模型}：证明合规不等于证明输出正确。\\
      应对：先明确只证明权限与流程合规，后续再叠加输出性质证明。
  \end{itemize}
\end{frame}
